const hello = (r, name, extra = {}) => ({ t: "hello", v: 1, token: tokenOf(name), board: r.ids.board, since: 0, client: name, ...extra });
const FAST = { limits: { hereMs: 20, herePublishMs: 100 } };
async function join(r, name, extra) { const s = await r.sock(); s.send(hello(r, name, extra)); await s.next(m => m.t === "welcome", `welcome ${name}`); return s; }
async function outcome(r, name, extra) { const s = await r.sock(); s.send(hello(r, name, extra)); await Promise.race([s.next(m => m.t === "welcome"), s.whenClosed()]); return s; }
const welcomedOf = s => s.msgs.some(m => m.t === "welcome");
async function viewersSettle(obs, n) {
  const last = () => obs.msgs.filter(m => m.t === "here").at(-1)?.viewers;
  const until = Date.now() + 10_000;
  while (last() !== n) { if (Date.now() > until) throw new Error(`viewers ${last()} want ${n}`); await sleep(10); }
}
/** 8 fresh tabs in, 9th refused; and 2 conns per a fresh tab then 3rd refused: proves writersBy/tabConns are empty for editor. */
async function capsClean(r, tag) {
  const open = [];
  for (let i = 0; i < 8; i++) open.push(await join(r, "editor", { tab: tag + i, client: tag + "c" + i }));
  const ninth = await outcome(r, "editor", { tab: tag + "9", client: tag + "c9" });
  expect(welcomedOf(ninth)).toBe(false);
  open.push(await join(r, "editor", { tab: tag + "0", client: tag + "d" }));
  const third = await outcome(r, "editor", { tab: tag + "0", client: tag + "e" });
  expect(welcomedOf(third)).toBe(false);
  await Promise.all(open.map(s => s.close())); await sleep(50);
}

    await sleep(60); for (const s of socks) s.ws.close();
  }
  await sleep(400); slow = false;
  await capsClean(r, "a");
  await r.stop();
});

test("token expiry closes free tabs and viewer counts", async () => {
  let expSoon = true;
  const r = await rig({ ...FAST, wrapAuth: a => ({ ...a, verify: async t => { const v = await a.verify(t); return v && expSoon && !t.includes("owner") ? { ...v, exp: Math.floor(Date.now() / 1000) + 2 } : v; } }) });
  const obs = await join(r, "owner");
  const socks = [];
  for (let i = 0; i < 8; i++) socks.push(await join(r, "editor", { tab: "x" + i, client: "x" + i }));
  socks.push(await join(r, "editor", { tab: "x0", client: "x0b" }));
  for (let i = 0; i < 3; i++) socks.push(await join(r, "viewer", { tab: i ? "vv" : undefined, client: "v" + i }));
  await viewersSettle(obs, 2);
  await Promise.all(socks.map(s => s.whenClosed()));
  expSoon = false;
  await viewersSettle(obs, 0);
  await capsClean(r, "c");
  await r.stop();
}, 20_000);

test("role-change close frees tabs", async () => {
  const r = await rig(FAST);
  const socks = [];
  for (let i = 0; i < 8; i++) socks.push(await join(r, "editor", { tab: "y" + i, client: "y" + i }));
  await r.registry.setGrant(r.ids.board, "dev:editor", "annotator");
  await r.bus.publish(subjects.grants(r.ids.tenant, r.ids.board), { board: r.ids.board }); await r.bus.flush?.();
  // trigger a grants recheck the way the registry announces it, if the rig doesn't: fall back to waiting for close
  const closed = await Promise.race([Promise.all(socks.map(s => s.whenClosed())).then(() => true), sleep(8000).then(() => false)]);
  if (!closed) { console.log("role-change close not triggered in rig; skipped"); await r.stop(); return; }
  await r.registry.setGrant(r.ids.board, "dev:editor", "editor"); await sleep(100);
  await capsClean(r, "d");
  await r.stop();
}, 20_000);

  const r = await rig({ verifyDelayMs: 80 });
  const socks = [];
  for (let i = 0; i < 12; i++) socks.push(await r.sock());
  socks.forEach((s, i) => s.send(hello(r, "editor", { tab: i < 3 ? "same" : "n" + i, client: "q" + i })));
  await Promise.all(socks.map(s => Promise.race([s.next(m => m.t === "welcome"), s.whenClosed()])));
  const inSame = socks.slice(0, 3).filter(welcomedOf).length;

});

test("MAPS: every churn path returns writersBy, tabConns, hvTabs to empty", async () => {
  let expSoon = false, slow = false, fail = false;
  const r = await rig({ ...FAST, wrapAuth: a => ({ ...a, verify: async t => { const v = await a.verify(t); return v && expSoon ? { ...v, exp: Math.floor(Date.now() / 1000) + 2 } : v; } }) });
  const open = r.server.rooms.open.bind(r.server.rooms);
  r.server.rooms.open = async (...a) => { if (fail) throw new Error("down"); if (slow) await sleep(150); return open(...a); };
  const settle = async () => { const until = Date.now() + 8000; while (JSON.stringify(maps()) !== JSON.stringify(zero) && Date.now() < until) await sleep(20); expect(maps()).toEqual(zero); };
  // open/close + refusals
  const s = [];
  for (let i = 0; i < 8; i++) s.push(await join(r, "editor", { tab: "o" + i, client: "o" + i }));
  s.push(await join(r, "editor", { tab: "o0", client: "o0b" }));
  await outcome(r, "editor", { tab: "o0", client: "o0c" }); await outcome(r, "editor", { tab: "o9", client: "o9" });
  for (let i = 0; i < 3; i++) s.push(await join(r, "viewer", { tab: i ? "vt" : undefined, client: "v" + i }));
  expect(maps()).toMatchObject({ writersBy: 1, tabConns: 8, hvTabs: 2, hv: 2 });
  await Promise.all(s.map(x => x.close())); await settle();
  // mid-hello close after first check
  slow = true;
  for (let i = 0; i < 6; i++) { const x = await r.sock(); x.send(hello(r, i % 2 ? "viewer" : "editor", { tab: "m" + (i % 3), client: "m" + i })); await sleep(40); x.ws.close(); }
  await sleep(400); slow = false; await settle();
  // rooms.open failing
  fail = true; for (let i = 0; i < 3; i++) await outcome(r, "editor", { tab: "f" + i, client: "f" + i }); fail = false; await settle();
  // token expiry
  expSoon = true; const e = [];
  for (let i = 0; i < 3; i++) e.push(await join(r, "editor", { tab: "e" + i, client: "e" + i }));
  e.push(await join(r, "viewer", { tab: "ev", client: "ev" }));
  expSoon = false;
  await Promise.all(e.map(x => x.whenClosed())); await settle();
  // role-change close
  const g = []; for (let i = 0; i < 3; i++) g.push(await join(r, "editor", { tab: "g" + i, client: "g" + i }));
  await r.registry.setGrant(r.ids.board, "dev:editor", "annotator");
  await r.bus.publish(subjects.grants(r.ids.tenant, r.ids.board), { board: r.ids.board }); await r.bus.flush?.();
  await Promise.all(g.map(x => x.whenClosed())); await settle();
  await r.stop();
}, 40_000);

test("SAME (user, tab) as viewer then writer (grant changed, no grants message)", async () => {
  const r = await rig(FAST);
  const obs = await join(r, "owner");
  const v = await join(r, "viewer", { tab: "X", client: "vx" });
  await viewersSettle(obs, 1);
  await r.registry.setGrant(r.ids.board, "dev:viewer", "editor");     // no grants message: the viewer conn stays
  const w1 = await join(r, "viewer", { tab: "X", client: "wx1" });      // now an editor on tab X
  const w2 = await join(r, "viewer", { tab: "X", client: "wx2" });      // 2nd writer conn on X: the viewer conn must not count
  const w3 = await outcome(r, "viewer", { tab: "X", client: "wx3" });
  expect(welcomedOf(w3)).toBe(false);
  expect(w3.msgs.at(-1)).toMatchObject({ code: "limit" });
  expect(maps()).toMatchObject({ writersBy: 2, tabConns: 2, hvTabs: 1, hv: 1 });   // owner observer + viewer|X writer tab
  await sleep(300);
  expect(obs.msgs.filter(m => m.t === "here").at(-1).viewers).toBe(1);
  for (const x of [v, w1, w2]) await x.close();
  await obs.close();
  const until = Date.now() + 5000; while (JSON.stringify(maps()) !== JSON.stringify(zero) && Date.now() < until) await sleep(20);
  expect(maps()).toEqual(zero);
  await r.stop();
}, 20_000);
