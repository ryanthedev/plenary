import { test, expect, afterAll } from "bun:test";

afterAll(cleanup);
const rect = (id, x = 0) => makeElement("rectangle", { id, x, y: 0, width: 10, height: 10 });
const hello = (r, name, extra = {}) => ({ t: "hello", v: 1, token: tokenOf(name), board: r.ids.board, since: 0, client: name, ...extra });
const op = (client, n, el) => ({ t: "op", id: `${client}:${n}`, sentAt: 0, stream: "board", op: { elements: [el] } });
async function readLog(store, r) { const out = []; for (;;) { const p = await store.read(r.ids.tenant, r.ids.board, out.length + 1, 5000); if (!p.length) break; out.push(...p); } return out; }

test("A1 revocation race: grants change while the hello waits on rooms.open", async () => {
  const gate = deferred();
  let gated = true;
  const r = await rig({ wrapStore: s => ({ ...s, openEpoch: async (...a) => { if (gated) await gate.promise; return s.openEpoch(...a); } }) });
  const s = await r.sock();
  s.send(hello(r, "editor"));
  await until(() => r.calls.role >= 1, "role checked");
  await sleep(100);                                    // now inside rooms.open (openEpoch gated)
  await r.registry.setGrant(r.ids.board, "dev:editor", null);
  await r.bus.publish(subjects.grants(r.ids.tenant, r.ids.board), { board: r.ids.board });
  await r.bus.flush(); await sleep(50);
  gated = false; gate.resolve();
  const first = await until(() => s.msgs[0] || s.closed, "an answer", 5000);
  console.log("A1 first answer:", JSON.stringify(first));
  if (s.msgs[0]?.t === "welcome") {
    s.send(op("editor", 1, rect("revoked-write")));
    const ack = await until(() => s.msgs.find(m => m.t === "ack" || m.t === "error") || s.closed, "op answer", 5000);
    console.log("A1 op answer after revocation:", JSON.stringify(ack));
    const log = await readLog(r.rawStore, r);
    console.log("A1 log ids:", log.map(e => `${e.id} by ${e.by}`));
  }
  await r.stop();
}, 20000);

test("A2 a real session token's expiry never closes a live connection (verify() returns no exp)", async () => {
  const sess = await makeSessions({ key: "k".repeat(40), ttlSec: 1 });
  const token = await sess.issue({ user: "dev:editor", name: "editor" });
  const r = await rig({ wrapAuth: a => ({ ...a, verify: async t => (t === token ? sess.verify(t) : a.verify(t)) }) });
  const s = await r.sock();
  s.send({ t: "hello", v: 1, token, board: r.ids.board, since: 0, client: "sx" });
  await s.next(m => m.t === "welcome");
  console.log("A2 identity has exp?", "exp" in (await sess.verify(token) ?? {}));
  await sleep(2500);                                   // the token is now past its exp
  console.log("A2 closed after expiry?", s.closed);
  if (!s.closed) {
    s.send(op("sx", 1, rect("after-expiry")));
    const a = await s.next(m => m.t === "ack" || m.t === "error", "answer", 5000);
    console.log("A2 op after expiry:", JSON.stringify(a));
  }
  await r.stop();
}, 20000);

test("A3 append durable but answered after the 4 s timeout: resend gets the same seq, logged once", async () => {
  let first = true;
  const r = await rig({ wrapStore: s => ({ ...s, append: async (...a) => { const res = await s.append(...a); if (first) { first = false; await sleep(4500); } return res; } }) });
  const s = await r.sock(); s.send(hello(r, "editor")); await s.next(m => m.t === "welcome");
  const m = op("editor", 1, rect("slow"));
  s.send(m);
  const ans = await s.next(x => x.t === "ack" || x.t === "error", "answer", 10000);
  console.log("A3 first answer:", JSON.stringify(ans), "closed:", JSON.stringify(await s.whenClosed(3000).catch(() => "open")));
  const t = await r.sock(); t.send(hello(r, "editor")); const w = await t.next(x => x.t === "welcome");
  console.log("A3 welcome after rebuild:", JSON.stringify(w));
  t.send(m);
  const a2 = await t.next(x => x.t === "ack" || x.t === "error", "resend answer", 10000);
  console.log("A3 resend answer:", JSON.stringify(a2));
  const log = await readLog(r.rawStore, r);
  expect(log.filter(e => e.id === "editor:1").length).toBe(1);
  expect(a2).toMatchObject({ t: "ack", seq: log.find(e => e.id === "editor:1").seq });
  await r.stop();
}, 30000);

test("A4 the write itself lands after the timeout (late append): logged once, ack matches the log", async () => {
  let first = true;
  const r = await rig({ wrapStore: s => ({ ...s, append: async (...a) => { if (first) { first = false; await sleep(4300); try { return await s.append(...a); } catch (e) { console.log("A4 late append:", e.code); throw e; } } return s.append(...a); } }) });
  const s = await r.sock(); s.send(hello(r, "editor")); await s.next(m => m.t === "welcome");
  const m = op("editor", 1, rect("late"));
  s.send(m);
  await s.next(x => x.t === "error", "unavailable", 10000);
  await s.whenClosed(3000).catch(() => {});
  await sleep(200);
  const t = await r.sock(); t.send(hello(r, "editor")); await t.next(x => x.t === "welcome");
  t.send(m);
  const a2 = await t.next(x => x.t === "ack" || x.t === "error", "resend answer", 10000);
  console.log("A4 resend answer:", JSON.stringify(a2));
  await sleep(500);
  const log = await readLog(r.rawStore, r);
  console.log("A4 log:", log.map(e => `${e.seq}:${e.id}`));
  expect(log.filter(e => e.id === "editor:1").length).toBe(1);
  if (a2.t === "ack") expect(a2.seq).toBe(log.find(e => e.id === "editor:1").seq);
  await r.stop();
}, 30000);

test("A5 two hosts over one Store: no seq reused, every acked op at its seq, none logged twice", async () => {
  const r1 = await rig();
  const r2 = await rig({ dir: r1.dir, registry: r1.registry, ids: r1.ids });
  const a = client(r1, "editor", "ca", { maxWaitMs: 50 }), b = client(r2, "editor0", "cb", { maxWaitMs: 50 });
  await a.connect(); await b.connect();
  await until(() => a.r.welcomed && b.r.welcomed, "both welcomed", 10000);
  for (let i = 0; i < 60; i++) { a.send("board", { elements: [rect("a" + i, i)] }); b.send("board", { elements: [rect("b" + i, i)] }); if (i % 10 === 0) await sleep(20); }
  await until(() => a.r.pending.size === 0 && b.r.pending.size === 0, "all ops settled", 60000);
  const acks = [...a.msgs, ...b.msgs].filter(m => m.t === "ack");
  const log = await readLog(r1.rawStore, r1);
  expect(log.map(e => e.seq)).toEqual(log.map((_, i) => i + 1));
  expect(new Set(log.map(e => `${e.by}|${e.id}`)).size).toBe(log.length);
  for (const k of acks) expect(log.find(e => e.id === k.id)?.seq).toBe(k.seq);
  console.log("A5 log length", log.length, "acks", acks.length, "errors", [...a.errors, ...b.errors].map(e => e.code), "connects", a.connects, b.connects);
  await a.drop(); await b.drop();
  await r2.stop(); await r1.stop();
}, 90000);

import { busBytes } from "../../server/bus/port.js";
test("A9 a 64 KB bus: big group commits split, a joiner's catch-up of ~1.5 MB never touches the bus", async () => {
  const CAP = 64 * 1024, refused = [], biggest = { n: 0 };
  const r = await rig({ wrapBus: b => ({ ...b, maxBytes: CAP, publish(s, m) { const n = busBytes(m); biggest.n = Math.max(biggest.n, n); if (n > CAP) { refused.push([s, n]); return Promise.reject(new Error("too-large")); } return b.publish(s, m); } }) });
  const w = await r.sock(); w.send(hello(r, "editor")); await w.next(m => m.t === "welcome");
  for (let i = 1; i <= 60; i++) w.send(op("editor", i, makeElement("text", { id: "t" + i, x: i, y: 0, width: 10, height: 10, text: "x".repeat(25_000) })));
  await w.next(m => m.t === "ack" && m.id === "editor:60", "last ack", 20000);
  await r.idle();
  const v = await r.sock(); v.send(hello(r, "viewer")); await v.next(m => m.t === "welcome");
  await until(() => { const s = new Set(); for (const m of v.msgs) { if (m.t === "entries") m.entries.forEach(e => s.add(e.seq)); if (m.t === "snapshot") for (let k = 1; k <= m.seq; k++) s.add(k); } return s.size >= 60; }, "viewer catch-up", 15000);
  console.log("A9 refused", refused, "biggest bus msg", biggest.n, "viewer frames", v.msgs.map(m => m.t).join(","));
  expect(refused).toEqual([]);
  await r.stop();
}, 60000);

test("R1 re-check passes stall the gateway's event loop in proportion to connections x grants messages", async () => {
  const { sqliteRegistry } = await import("../../server/registry/sqlite.js");
  const { tempDir } = await import("./helpers.js");
  const registry = sqliteRegistry({ path: tempDir() + "/registry.db" });
  const r = await rig({ registry, limits: { maxConnsPerUser: 100000 } });
  const N = 2000, socks = [];
  for (let i = 0; i < N; i += 100) {
    const batch = await Promise.all(Array.from({ length: 100 }, () => r.sock()));
    batch.forEach((s, k) => s.send({ t: "hello", v: 1, token: tokenOf("viewer"), board: r.ids.board, since: 0, client: "v" + (i + k) }));
    socks.push(...batch);
  }
  await until(() => socks.every(s => s.msgs.some(m => m.t === "welcome")), "all welcomed", 60000);
  const measure = async (label, fire) => {
    let maxLag = 0, last = performance.now();
    const probe = setInterval(() => { const n = performance.now(); maxLag = Math.max(maxLag, n - last - 2); last = n; }, 2);
    await sleep(50); fire(); await sleep(1500); clearInterval(probe);
    console.log(`R1 ${label}: max event-loop stall ${Math.round(maxLag)} ms`);
  };
  await measure("baseline", () => {});
  await measure("1 grants message", () => r.bus.publish(subjects.grants(r.ids.tenant, r.ids.board), { board: r.ids.board }));
  await measure("20 grants messages", () => { for (let i = 0; i < 20; i++) r.bus.publish(subjects.grants(r.ids.tenant, r.ids.board), { board: r.ids.board }); });
  for (const s of socks) s.ws.close();
  await r.stop(); await registry.close();
}, 180000);

test("R2 a short Registry blip during the timed re-check closes every live connection on the gateway", async () => {
  let down = false, roleCalls = 0;
  const r = await rig({ limits: { maxConnsPerUser: 100000, roleRecheckMs: 1000 }, wrapAuth: a => ({ ...a, role: async (id, b) => { roleCalls++; if (down) throw new Error("registry blip"); return a.role(id, b); } }) });
  const N = 300, socks = await Promise.all(Array.from({ length: N }, () => r.sock()));
  socks.forEach((s, i) => s.send({ t: "hello", v: 1, token: tokenOf(i % 10 ? "viewer" : "editor" + (i % 8)), board: r.ids.board, since: 0, client: "c" + i }));
  await until(() => socks.every(s => s.msgs.some(m => m.t === "welcome")), "all welcomed", 30000);
  await sleep(1100);
  const before = socks.filter(s => s.closed).length;
  roleCalls = 0; down = true; await sleep(1100); down = false; await sleep(200);
  const closed = socks.filter(s => s.closed);
  console.log(`R2 closed before blip ${before}/${N}; by a ~1 s blip ${closed.length}/${N}; role calls during blip ${roleCalls}`);
  await r.stop();
}, 60000);
